Find DOM Elements by Color

This question is about traversing the DOM tree and collecting every element whose effective color matches a given color value.

Problem Summary

Given a root element and a color like red, find all descendant elements whose computed color matches that color.

  • Start from a parent element.
  • Visit each child recursively.
  • Check the current element's computed style.
  • Collect elements whose color matches the target.

How the solution works

  1. getComputedColorCode(colorCode) creates a temporary element and applies the given color to it. This lets the browser normalize the value to a standard format like rgb(255, 0, 0).
  2. findElementByColor(root, colorCode) starts with the given root and recursively walks every descendant.
  3. For each element, it reads the computed color using window.getComputedStyle(node).color and compares it to the target color.
  4. When a match is found, the element is pushed into an array that is returned at the end.

Implementation

function getComputedColorCode(colorCode) {
  const temp = document.createElement('div');
  temp.style.color = colorCode;
  document.body.appendChild(temp);
  const computedColor = window.getComputedStyle(temp).color;
  document.body.removeChild(temp);
  return computedColor;
}

function findElementByColor(root, colorCode) {
  const targetColor = getComputedColorCode(colorCode);
  const output = [];

  function search(node) {
    if (!node || node.nodeType !== Node.ELEMENT_NODE) return;

    if (window.getComputedStyle(node).color === targetColor) {
      output.push(node);
    }

    for (const child of node.children) {
      search(child);
    }
  }

  search(root);
  return output;
}
