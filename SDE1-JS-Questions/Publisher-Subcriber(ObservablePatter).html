Publisher-Subscriber (Observable Pattern)

This document explains a simple observable implementation where subscribers receive emitted values.

Subscribers can unsubscribe at any time to stop receiving notifications.

Implementation

function Observable() {
  this.observers = [];

  this.subscribe = (callback) => {
    if (typeof callback !== 'function') {
      throw new Error('callback should be a function');
    }
    const observer = { callback };
    this.observers.push(observer);
    return {
      unsubscribe: () => {
        const index = this.observers.indexOf(observer);
        if (index > -1) {
          this.observers.splice(index, 1);
        }
      },
    };
  };

  this.notify = (...data) => {
    this.observers.forEach((observer) => {
      try {
        observer.callback(...data);
      } catch (err) {
        console.error('Error while invoking callback', err);
      }
    });
  };
}
