Remove Cycle From Object

This page explains how to break a cyclic object reference using recursion and a WeakSet.

A cycle is removed by replacing repeated references with null.

Implementation

function removeCycle(obj) {
  const set = new WeakSet([obj]);

  function iterateObj(current) {
    for (let key in current) {
      if (!current.hasOwnProperty(key)) continue;
      const value = current[key];
      if (typeof value === 'object' && value !== null) {
        if (set.has(value)) {
          current[key] = null;
        } else {
          set.add(value);
          iterateObj(value);
        }
      }
    }
  }

  iterateObj(obj);
}
