Pipe Object Values

This example evaluates functions inside an object and replaces them with their returned values.

The Pipe helper recursively traverses nested object values.

Implementation

function Pipe(obj) {
  return (...args) => {
    for (let key in obj) {
      const value = obj[key];
      if (typeof value === 'function') {
        obj[key] = value(...args);
      } else if (typeof value === 'object' && value !== null) {
        obj[key] = Pipe(value)(...args);
      } else {
        obj[key] = value;
      }
    }
    return obj;
  };
}
